cmake_minimum_required(VERSION 3.22)
project(sayin_helper VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(helper_core STATIC src/parcel.cpp src/history.cpp src/app_protocol.cpp)
target_include_directories(helper_core PUBLIC include)
if(MSVC)
    target_compile_options(helper_core PRIVATE /W4 /WX)
else()
    target_compile_options(helper_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ANDROID)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS module/app-policy.json)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/module/app-policy.json" POLICY)
    string(JSON PIN GET "${POLICY}" allowed_signing_certificate_sha256 0)
    string(JSON APP GET "${POLICY}" application_ids 0)
    string(JSON TEST_APP GET "${POLICY}" application_ids 1)
    configure_file(include/app_policy.hpp.in generated/app_policy.hpp @ONLY)
    add_executable(sayin-helper src/main.cpp src/binder.cpp src/storage.cpp src/crypto.cpp src/app_bridge.cpp)
    target_include_directories(sayin-helper PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(sayin-helper PRIVATE helper_core binder_ndk dl)
    target_compile_options(sayin-helper PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
    target_link_options(sayin-helper PRIVATE -Wl,-z,relro,-z,now)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(helper_core_test tests/core_test.cpp)
    target_link_libraries(helper_core_test PRIVATE helper_core)
    add_test(NAME helper_core COMMAND helper_core_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
endif()
