cmake_minimum_required(VERSION 3.22.1)
project(sayin_translation LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
if (SAYIN_R2T2_BENCHMARK_BACKEND STREQUAL "arm")
    set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
elseif (SAYIN_R2T2_BENCHMARK_BACKEND STREQUAL "vulkan")
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
elseif (ANDROID_ABI STREQUAL "arm64-v8a" AND NOT SAYIN_R2T2_BENCHMARK_BACKEND)
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
    # The FP16 ASR projector benefits from ggml's tiled matrix multiply on ARMv8.2.
    set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
endif ()
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMIT "32bdc3a89e6528d3e419024c833c5dabf32c7541" CACHE STRING "" FORCE)
set(LLAMA_BUILD_NUMBER 0 CACHE STRING "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../local-translation/llama.cpp")
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
set(MTMD_SOURCE "${LLAMA_SOURCE}/tools/mtmd/mtmd.cpp")
# Build the R2T2 extensions in mtmd's translation unit without changing the pinned source.
set(MTMD_EXTENDED "${CMAKE_CURRENT_BINARY_DIR}/mtmd_sayin.cpp")
file(CONFIGURE OUTPUT "${MTMD_EXTENDED}" CONTENT
    "#include \"@MTMD_SOURCE@\"\n#include \"@CMAKE_CURRENT_LIST_DIR@/r2t2_mtmd.inc\"\n" @ONLY)
get_target_property(MTMD_SOURCES mtmd SOURCES)
if (NOT "mtmd.cpp" IN_LIST MTMD_SOURCES)
    message(FATAL_ERROR "The pinned mtmd source list has changed")
endif ()
list(REMOVE_ITEM MTMD_SOURCES mtmd.cpp)
set_property(TARGET mtmd PROPERTY SOURCES "${MTMD_SOURCES};${MTMD_EXTENDED}")
if (GGML_VULKAN)
    find_package(SPIRV-Headers CONFIG REQUIRED)
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif ()
add_library(sayin_translation SHARED translation_jni.cpp r2t2_jni.cpp)
if (SAYIN_R2T2_BENCHMARK_BACKEND STREQUAL "vulkan")
    target_compile_definitions(sayin_translation PRIVATE SAYIN_R2T2_VULKAN_BENCHMARK)
endif ()
if (GGML_CPU_ALL_VARIANTS)
    target_compile_definitions(sayin_translation PRIVATE SAYIN_ANDROID_CPU_VARIANTS)
endif ()
target_link_libraries(sayin_translation PRIVATE mtmd llama)
target_compile_options(sayin_translation PRIVATE -Wall -Wextra -Werror)
target_link_options(sayin_translation PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
